Python request example

Replace HOST_IP with the address of this server, then run the example with Python. 

import json
from urllib.request import Request, urlopen

base_url = "http://HOST_IP:8000"
payload = {
    "model": "d1-omni-2b",
    "state": (
        "You are Chromite, a small tadpole creature.\n\n"
        "User message: The pet owner says hello."
    ),
    "questions": {
        "greeting": {
            "type": "noul",
            "instructions": "Is the pet owner greeting Chromite?",
        },
    },
}
request = Request(
    f"{base_url}/v1/systemone",
    data=json.dumps(payload).encode("utf-8"),
    headers={"Content-Type": "application/json"},
)
with urlopen(request) as response:
    print(json.dumps(json.load(response), indent=2))
